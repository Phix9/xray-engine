cmake_minimum_required (VERSION 3.10)

project ("XrayEngine")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(PROJECT_SOURCE_DIR STREQUAL PROJECT_BINARY_DIR)
  message(
    FATAL_ERROR
      "In-source builds not allowed. Please make a new directory (called a build directory) and run CMake from there."
  )
endif()

set(ENGINE_ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

find_package(Vulkan REQUIRED)

add_subdirectory(shader)

file(GLOB_RECURSE HEADER_FILES "*.hpp")
file(GLOB_RECURSE SOURCE_FILES "*.cpp")

list(FILTER HEADER_FILES EXCLUDE REGEX "/build/")
list(FILTER SOURCE_FILES EXCLUDE REGEX "/build/")

source_group(TREE "${ENGINE_ROOT_DIR}" FILES ${HEADER_FILES} ${SOURCE_FILES})

add_executable(XrayEngine ${HEADER_FILES} ${SOURCE_FILES})

set(includes $ENV{includes})
set(GLFW_DIR $ENV{glfw_ROOT})

target_include_directories(XrayEngine PUBLIC
    "${ENGINE_ROOT_DIR}"
    "${ENGINE_ROOT_DIR}/shader/generated/cpp"
    "${includes}"
)

target_link_libraries(XrayEngine PRIVATE
    "${GLFW_DIR}/lib-vc2022/glfw3.lib"
    Vulkan::Vulkan
)
